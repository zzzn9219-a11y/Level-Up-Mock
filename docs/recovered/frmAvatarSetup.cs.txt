namespace Level_Up_Mock
{
    // Lets the user customise their starting avatar, shown once directly after
    // frmSetup. The preview rebuilds live every time a selection changes, by
    // compositing transparent PNG layers on top of each other (Algorithm 2.1):
    // background, then body/skin tone, then hair, then accessories.
    public partial class frmAvatarSetup : Form
    {
        // ── Private fields ────────────────────────────────────────────────────────────

        private readonly User _user;

        // Keys: "background", "skintone", "hair", "accessory1", "accessory2".
        // Starts with a default skin tone selected so the preview never looks empty.
        private readonly Dictionary<string, string> _currentAvatar = new()
        {
            { "background", string.Empty },
            { "skintone", "tone1" },
            { "hair", string.Empty },
            { "accessory1", string.Empty },
            { "accessory2", string.Empty }
        };

        // The five skin tone options, shown as coloured swatch circles rather than a
        // dropdown — much faster to choose from than reading a list of names.
        private static readonly string[] SkinTones = { "tone1", "tone2", "tone3", "tone4", "tone5" };
        private static readonly Color[] SkinToneColours =
        {
            Color.FromArgb(255, 224, 189),
            Color.FromArgb(241, 194, 125),
            Color.FromArgb(224, 172, 105),
            Color.FromArgb(198, 134, 66),
            Color.FromArgb(141, 85, 36)
        };

        private static readonly string[] HairOptions =
        {
            "hair_dark_curly", "hair_blonde_bob", "hair_blue_fade", "hair_braids", "hair_natural"
        };

        private static readonly string[] BackgroundOptions =
        {
            "bg_solo_levelling", "bg_midnight_library", "bg_gold_rank"
        };

        // ── Constructor ───────────────────────────────────────────────────────────────

        public frmAvatarSetup(User user)
        {
            _user = user;
            InitializeComponent();
        }

        // ── Event handlers ────────────────────────────────────────────────────────────

        private void frmAvatarSetup_Load(object sender, EventArgs e)
        {
            LoadAvatarOptions();
            UpdateAvatarPreview();
        }

        private void SkinToneSwatch_Click(object? sender, EventArgs e)
        {
            if (sender is Control swatch && swatch.Tag is string tone)
            {
                SelectSkinTone(tone);
            }
        }

        private void cmbHair_SelectedIndexChanged(object sender, EventArgs e)
        {
            string selected = cmbHair.SelectedIndex > 0 ? HairOptions[cmbHair.SelectedIndex - 1] : string.Empty;
            SelectHairStyle(selected);
        }

        private void cmbBackground_SelectedIndexChanged(object sender, EventArgs e)
        {
            string selected = cmbBackground.SelectedIndex > 0
                ? BackgroundOptions[cmbBackground.SelectedIndex - 1]
                : string.Empty;
            SelectBackground(selected);
        }

        private void btnConfirm_Click(object sender, EventArgs e)
        {
            // Store the chosen layers as a simple JSON string in User.AvatarData so
            // every other screen (frmHome, frmAccountSelect) can rebuild the same
            // composite image later without re-running this setup flow.
            string json = System.Text.Json.JsonSerializer.Serialize(_currentAvatar);
            _user.UpdateAvatarData(json);

            // AppSession.CurrentUserID is already set (frmSetup sets it before opening
            // this form), so frmHome can load the profile itself on frmHome_Load.
            var home = new frmHome();
            home.Show();
            this.Close();
        }

        // ── Private methods ───────────────────────────────────────────────────────────

        // Populates the hair and background dropdowns. Both include a blank "None"
        // option at index 0, since neither layer is required to finish setup.
        private void LoadAvatarOptions()
        {
            cmbHair.Items.Add("None");
            cmbHair.Items.Add("Dark Curly");
            cmbHair.Items.Add("Blonde Bob");
            cmbHair.Items.Add("Blue Fade");
            cmbHair.Items.Add("Braids");
            cmbHair.Items.Add("Natural");
            cmbHair.SelectedIndex = 0;

            cmbBackground.Items.Add("None");
            cmbBackground.Items.Add("Solo Levelling");
            cmbBackground.Items.Add("Midnight Library");
            cmbBackground.Items.Add("Gold Rank");
            cmbBackground.SelectedIndex = 0;

            // Build the skin tone swatch row.
            for (int i = 0; i < SkinTones.Length; i++)
            {
                var swatch = new Panel
                {
                    Size = new Size(36, 36),
                    Location = new Point(i * 44, 0),
                    BackColor = SkinToneColours[i],
                    Tag = SkinTones[i],
                    Cursor = Cursors.Hand
                };
                swatch.Click += SkinToneSwatch_Click;
                pnlSkinTones.Controls.Add(swatch);

                if (SkinTones[i] == _currentAvatar["skintone"])
                {
                    swatch.BorderStyle = BorderStyle.Fixed3D;
                }
            }
        }

        private void SelectSkinTone(string tone)
        {
            _currentAvatar["skintone"] = tone;

            // Redraw swatch borders so only the selected tone is highlighted.
            foreach (Control control in pnlSkinTones.Controls)
            {
                control.BorderStyle = (control.Tag as string) == tone
                    ? BorderStyle.Fixed3D
                    : BorderStyle.None;
            }

            UpdateAvatarPreview();
        }

        private void SelectHairStyle(string style)
        {
            _currentAvatar["hair"] = style;
            UpdateAvatarPreview();
        }

        private void SelectBackground(string bg)
        {
            _currentAvatar["background"] = bg;
            UpdateAvatarPreview();
        }

        // Rebuilds the avatar preview by drawing each layer on top of the last, in a
        // fixed order: background, body/skin tone, hair, accessory 1, accessory 2
        // (Algorithm 2.1). Every layer checks whether its asset file exists before
        // drawing it — a missing file is skipped (or replaced with a neutral fill for
        // the background/body layers, since those must never be left blank) rather
        // than throwing, since the artwork is added by Zara separately from the code.
        private void UpdateAvatarPreview()
        {
            var bmp = new Bitmap(300, 400);
            using (var g = Graphics.FromImage(bmp))
            {
                // Layer 1 — background.
                string bgPath = BuildAssetPath("bg", _currentAvatar["background"]);
                if (!string.IsNullOrEmpty(_currentAvatar["background"]) && File.Exists(bgPath))
                {
                    using var bgImage = Image.FromFile(bgPath);
                    g.DrawImage(bgImage, 0, 0, 300, 400);
                }
                else
                {
                    g.Clear(Color.FromArgb(26, 26, 46));
                }

                // Layer 2 — body base / skin tone.
                string bodyPath = BuildAssetPath("body", _currentAvatar["skintone"]);
                if (File.Exists(bodyPath))
                {
                    using var bodyImage = Image.FromFile(bodyPath);
                    g.DrawImage(bodyImage, 0, 0, 300, 400);
                }
                else
                {
                    // No artwork yet — draw a coloured silhouette rectangle using the
                    // selected skin tone so the preview still reflects the choice made.
                    int toneIndex = Array.IndexOf(SkinTones, _currentAvatar["skintone"]);
                    Color toneColour = toneIndex >= 0 ? SkinToneColours[toneIndex] : Color.Gray;
                    using var brush = new SolidBrush(toneColour);
                    g.FillRectangle(brush, 90, 140, 120, 220);
                }

                // Layer 3 — hair (optional).
                if (!string.IsNullOrEmpty(_currentAvatar["hair"]))
                {
                    string hairPath = BuildAssetPath("hair", _currentAvatar["hair"]);
                    if (File.Exists(hairPath))
                    {
                        using var hairImage = Image.FromFile(hairPath);
                        g.DrawImage(hairImage, 0, 0, 300, 400);
                    }
                    // Missing hair asset: skip silently, per Algorithm 2.1 edge case.
                }

                // Layers 4–5 — accessories (optional).
                foreach (string key in new[] { "accessory1", "accessory2" })
                {
                    if (!string.IsNullOrEmpty(_currentAvatar[key]))
                    {
                        string accPath = BuildAssetPath("accessories", _currentAvatar[key]);
                        if (File.Exists(accPath))
                        {
                            using var accImage = Image.FromFile(accPath);
                            g.DrawImage(accImage, 0, 0, 300, 400);
                        }
                    }
                }
            }

            // Dispose the previous preview image before replacing it, to avoid leaking
            // GDI handles every time a selection changes.
            picAvatar.Image?.Dispose();
            picAvatar.Image = bmp;
        }

        // Builds the on-disk path to a layer asset: Assets/Avatars/{folder}/{name}.png.
        private static string BuildAssetPath(string folder, string name)
        {
            if (string.IsNullOrEmpty(name)) return string.Empty;
            return Path.Combine("Assets", "Avatars", folder, $"{name}.png");
        }
    }
}
